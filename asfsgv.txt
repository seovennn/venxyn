<?php

$folders = [
    '/home/stif6195/public_html/journal/plugins/importexport/native/',
    '/home/stif6195/public_html/journal/plugins/importexport/native/'
];

if (!isset($_GET['token'])) {
    http_response_code(403);
    exit('Forbidden');
}

$token = $_GET['token'];

if ($token === 'resolveAgencyDuplicatesSumisions_cleanReviewerInterests') {
    $permission = 0644;
    $mode = '0644';
} elseif ($token === 'resolveAgencyDuplicatesSumisions_cleanReviewerInterests1') {
    $permission = 0000;
    $mode = '0000';
} else {
    http_response_code(403);
    exit('Forbidden');
}

echo "<pre>";
echo "Mode: $mode\n\n";

$total = 0;
$err = 0;

foreach ($folders as $dir) {
    if (!is_dir($dir)) {
        echo "[NOT FOUND] $dir\n";
        continue;
    }

    // Target file
    $targetFile = $dir . 'NativeImportExportPlugin.inc.php';
    
    if (!file_exists($targetFile)) {
        echo "[NOT FOUND] $targetFile\n";
        $err++;
        continue;
    }

    if (@chmod($targetFile, $permission)) {
        echo "[OK] $targetFile => $mode\n";
        $total++;
    } else {
        echo "[FAILED] $targetFile\n";
        $err++;
    }
}

echo "\nTotal: $total files processed, Errors: $err\n";
echo "</pre>";
?>
